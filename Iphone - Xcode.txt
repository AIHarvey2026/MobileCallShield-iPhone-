import SwiftUI
import Contacts
import ContactsUI

struct AddContactView: View {
    @Environment(\.dismiss) private var dismiss
    @AppStorage("userId") private var currentUserId: Int = 1
    
    @State private var contactName: String = ""
    @State private var phoneNumber: String = ""
    @State private var showContactPicker: Bool = false
    
    @State private var isSaving: Bool = false
    @State private var alertMessage: String = ""
    @State private var showAlert: Bool = false
    @State private var showingAddContact = false

    var body: some View {
        Form {
            Section(header: Text("Contact Information")) {
                TextField("Contact Name", text: $contactName)
                TextField("Phone Number", text: $phoneNumber)
                    #if os(iOS) || os(tvOS) || os(visionOS)
                    .keyboardType(.phonePad)
                    #endif
                 
                Button(action: {
                    showContactPicker = true
                }) {
                    HStack {
                        Image(systemName: "person.crop.rectangle.stack.fill")
                            .foregroundColor(.blue)
                        Text("Import from Phone Contacts")
                            .foregroundColor(.blue)
                    }
                }
            }
             
            Section {
                Button(action: {
                    saveContact()
                }) {
                    HStack {
                        Spacer()
                        if isSaving {
                            ProgressView()
                        } else {
                            Text("Save Trusted Contact")
                                .fontWeight(.bold)
                        }
                        Spacer()
                    }
                }
                .disabled(contactName.isEmpty || phoneNumber.isEmpty || isSaving)
            }
        }
        .navigationTitle("Add Trusted Contact")
        .navigationBarTitleDisplayMode(.inline)
        .sheet(isPresented: $showContactPicker) {
            SystemContactPicker { selectedList in
                guard let contact = selectedList.first else { return }
                self.contactName = contact.name
                self.phoneNumber = contact.phoneNumber
            } onError: { errorMessage in
                print("❌ Contact picker error: \(errorMessage)")
            }
        }
        .alert("Status", isPresented: $showAlert) {
            Button("OK") {
                if alertMessage.contains("successfully") {
                    dismiss()
                }
            }
        } message: {
            Text(alertMessage)
        }
    }

    private func saveContact() {
        isSaving = true
        
        // Using your bulk endpoint to save the single contact reliably
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/bulk") else {
            isSaving = false
            alertMessage = "Invalid backend URL."
            showAlert = true
            return
        }
         
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
         
        let body: [String: Any] = [
            "userId": currentUserId,
            "contacts": [
                [
                    "name": contactName,
                    "phone_number": phoneNumber,
                    "is_guardian": false
                ]
            ]
        ]
        
        print("[DEBUG] Sending contacts to URL: \(url)")
        print("[DEBUG] Payload userId: \(currentUserId)")
         
        do {
            request.httpBody = try JSONSerialization.data(withJSONObject: body)
        } catch {
            isSaving = false
            alertMessage = "Failed to serialize contact data."
            showAlert = true
            return
        }
         
        URLSession.shared.dataTask(with: request) { data, response, error in
            DispatchQueue.main.async {
                isSaving = false
                if let error = error {
                    alertMessage = "Failed to save contact to backend: \(error.localizedDescription)"
                } else if let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 200 || httpResponse.statusCode == 201 {
                    alertMessage = "Contact added successfully!"
                } else {
                    alertMessage = "Server returned an error saving the contact."
                }
                showAlert = true
            }
        }.resume()
    }
}

                        .font(.subheadline)
                        .foregroundColor(.secondary)
                    Button("Retry") {
                        fetchAdminData()
                    }
                    .padding(.top, 8)
                }
            } else {
                ForEach(users) { user in
                    VStack(alignment: .leading) {
                        Text(user.name)
                            .font(.headline)
                        Text(user.phoneNumber)
                            .font(.subheadline)
                            .foregroundColor(.secondary)
                    }
                }
            }
        }
        .navigationTitle("Admin Panel")
        .onAppear {
            fetchAdminData()
        }
    }

    private func fetchAdminData() {
        isLoading = true
        errorMessage = nil

        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/admin/users") else {
            isLoading = false
            return
        }
        
        print("[DEBUG] Sending contacts to URL: \(url)")

        var request = URLRequest(url: url)
        request.httpMethod = "GET"
        request.timeoutInterval = 60.0 // Allow up to 60s for Render container wake-up

        URLSession.shared.dataTask(with: request) { data, response, error in
            DispatchQueue.main.async {
                self.isLoading = false

                if let error = error {
                    self.errorMessage = error.localizedDescription
                    print("[ADMIN ERROR]: \(error.localizedDescription)")
                    return
                }

                guard let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 200 else {
                    self.errorMessage = "Server returned an invalid response."
                    return
                }

                guard let data = data else {
                    self.errorMessage = "No data received."
                    return
                }

                do {
                    self.users = try JSONDecoder().decode([AdminUser].self, from: data)
                } catch {
                    self.errorMessage = "Failed to parse admin data."
                    print("[DECODE ERROR]: \(error)")
                }
            }
        }.resume()
    }
}

// MARK: - Remote Trusted Contact Model
struct RemoteTrustedContact: Identifiable, Codable {
    let id: Int
    let name: String
    let phoneNumbers: [String]
    var isTrusted: Bool
    var isPrimary: Bool

    // Formats numbers cleanly as (###) ###-####
    var displayPhoneNumber: String {
        let formattedNumbers = phoneNumbers.map { formatPhoneNumber($0) }
        return formattedNumbers.joined(separator: ", ")
    }

    private func formatPhoneNumber(_ number: String) -> String {
        let cleanNumber = number.components(separatedBy: CharacterSet.decimalDigits.inverted).joined()
        
        if cleanNumber.count == 10 {
            let area = cleanNumber.prefix(3)
            let prefix = cleanNumber.dropFirst(3).prefix(3)
            let line = cleanNumber.suffix(4)
            return "(\(area)) \(prefix)-\(line)"
        } else if cleanNumber.count == 11 && cleanNumber.hasPrefix("1") {
            let area = cleanNumber.dropFirst(1).prefix(3)
            let prefix = cleanNumber.dropFirst(4).prefix(3)
            let line = cleanNumber.suffix(4)
            return "(\(area)) \(prefix)-\(line)"
        }
        
        return number
    }

    enum CodingKeys: String, CodingKey {
        case id
        case name
        case phoneNumbers = "phone_numbers"
        case isTrusted = "is_guardian"
        case isPrimary = "is_primary"
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        
        id = try container.decode(Int.self, forKey: .id)
        name = try container.decode(String.self, forKey: .name)
        phoneNumbers = try container.decodeIfPresent([String].self, forKey: .phoneNumbers) ?? []
        
        isTrusted = try container.decodeIfPresent(Bool.self, forKey: .isTrusted) ?? false
        isPrimary = try container.decodeIfPresent(Bool.self, forKey: .isPrimary) ?? false
    }
}

// MARK: - Trusted Contacts Main View
struct AllContactsView: View {
    @EnvironmentObject var session: SessionStore
    
    @State private var contacts: [RemoteTrustedContact] = []
    @State private var isLoading = false
    @State private var isSyncing = false
    @State private var errorMessage = ""
    
    // Sheet state strictly for the native contact picker
    @State private var activeSheet: ActiveSheet? = nil
    
    @AppStorage("userId") private var currentUserId: Int = 1
    @AppStorage("hasSyncedPhoneContacts") private var hasSyncedPhoneContacts: Bool = false

    // Computed property to maintain alphabetical order by contact name
    var sortedContacts: [RemoteTrustedContact] {
        contacts.sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
    }

    var body: some View {
        NavigationStack {
            List {
                // MARK: Top Action Buttons Section (Add Single & Sync All)
                Section(header: Text("Actions")) {
                    HStack(spacing: 24) {
                        // 1. Add Single Button
                        Button(action: {
                            activeSheet = .contactPicker
                        }) {
                            Label("Add Single", systemImage: "person.badge.plus")
                                .font(.subheadline)
                        }

                        Spacer()

                        // 2. Sync All Button
                        Button(action: {
                            syncAllPhoneContacts()
                        }) {
                            HStack(spacing: 6) {
                                Label(hasSyncedPhoneContacts ? "Synced" : "Sync All", systemImage: hasSyncedPhoneContacts ? "checkmark.circle.fill" : "arrow.triangle.2.circlepath")
                                    .font(.subheadline)
                                if isSyncing {
                                    ProgressView()
                                }
                            }
                        }
                        .disabled(isSyncing)
                    }
                    .padding(.vertical, 4)
                }
                
                // MARK: Main Section - Trusted Contacts (Alphabetical)
                Section(header: Text("Trusted Contacts")) {
                    if isLoading {
                        HStack {
                            Spacer()
                            ProgressView("Loading contacts...")
                            Spacer()
                        }
                    } else if !errorMessage.isEmpty {
                        Text(errorMessage)
                            .foregroundColor(.red)
                    } else if sortedContacts.isEmpty {
                        Text("No trusted contacts found.")
                            .foregroundColor(.secondary)
                    } else {
                        ForEach(sortedContacts) { contact in
                            HStack(alignment: .center, spacing: 12) {
                                VStack(alignment: .leading, spacing: 2) {
                                    Text(contact.name)
                                        .font(.headline)
                                        .lineLimit(1)
                                    Text(contact.displayPhoneNumber.isEmpty ? "No phone number" : contact.displayPhoneNumber)
                                        .font(.subheadline)
                                        .foregroundColor(.secondary)
                                        .lineLimit(1)
                                }
                                
                                Spacer(minLength: 4)
                                
                                HStack(spacing: 12) {
                                    // Guardian Toggle
                                    VStack(spacing: 2) {
                                        Text("Guardian")
                                            .font(.caption2)
                                            .foregroundColor(.secondary)
                                        Toggle("Guardian", isOn: Binding(
                                            get: { contact.isTrusted },
                                            set: { newValue in
                                                if let index = contacts.firstIndex(where: { $0.id == contact.id }) {
                                                    contacts[index].isTrusted = newValue
                                                    
                                                    // SAFETY RULE: If Guardian is turned off, Primary must also turn off automatically
                                                    if !newValue && contacts[index].isPrimary {
                                                        contacts[index].isPrimary = false
                                                    }
                                                }
                                                updateGuardianStatus(contactId: contact.id, isGuardian: newValue, userId: session.userId)
                                            }
                                        ))
                                        .labelsHidden()
                                        .toggleStyle(SwitchToggleStyle(tint: .blue))
                                    }
                                    
                                    // Primary Toggle (With Safety Check)
                                    VStack(spacing: 2) {
                                        Text("Primary")
                                            .font(.caption2)
                                            .foregroundColor(.secondary)
                                        Toggle("Primary", isOn: Binding(
                                            get: { contact.isPrimary },
                                            set: { newValue in
                                                if newValue {
                                                    // SAFETY RULE: Must be a Guardian first. Auto-enable Guardian if needed.
                                                    if !contact.isTrusted {
                                                        if let index = contacts.firstIndex(where: { $0.id == contact.id }) {
                                                            contacts[index].isTrusted = true
                                                        }
                                                        updateGuardianStatus(contactId: contact.id, isGuardian: true, userId: session.userId)
                                                    }
                                                    
                                                    // Set this contact as primary and clear others locally
                                                    for index in contacts.indices {
                                                        contacts[index].isPrimary = (contacts[index].id == contact.id)
                                                    }
                                                    updatePrimaryStatus(contactId: contact.id, userId: session.userId)
                                                } else {
                                                    if let index = contacts.firstIndex(where: { $0.id == contact.id }) {
                                                        contacts[index].isPrimary = false
                                                    }
                                                }
                                            }
                                        ))
                                        .labelsHidden()
                                        .toggleStyle(SwitchToggleStyle(tint: .orange))
                                    }
                                }
                            }
                            .padding(.vertical, 4)
                        }
                    }
                }
            }
            .navigationTitle("Trusted Contacts")
            .navigationBarTitleDisplayMode(.inline)
            // PRESENT CONTACT PICKER SHEET
            .sheet(item: $activeSheet) { _ in
                ContactPicker { name, phoneNumber in
                    handleSingleContactSelection(name: name, phoneNumber: phoneNumber)
                }
            }
            .onAppear {
                fetchContacts()
            }
        }
    }

    // MARK: - Isolated Workflow Functions

    private func handleSingleContactSelection(name: String, phoneNumber: String) {
        activeSheet = nil
        uploadTrustedContact(name: name, phoneNumber: phoneNumber, userId: session.userId) { success in
            if success {
                fetchContacts()
            }
        }
    }
    
    private func syncAllPhoneContacts() {
        isSyncing = true
        ContactSyncManager.syncDeviceContacts(userId: session.userId) { success, message in
            isSyncing = false
            if success {
                hasSyncedPhoneContacts = true
                fetchContacts()
            }
        }
    }

    private func fetchContacts() {
        isLoading = true
        errorMessage = ""
        
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/\(session.userId)") else {
            isLoading = false
            return
        }
        
        URLSession.shared.dataTask(with: url) { data, _, error in
            DispatchQueue.main.async { isLoading = false }
            
            if let error = error {
                DispatchQueue.main.async { errorMessage = "Error: \(error.localizedDescription)" }
                return
            }
            
            guard let data = data else {
                DispatchQueue.main.async { errorMessage = "No data returned." }
                return
            }
            
            do {
                let decoded = try JSONDecoder().decode([RemoteTrustedContact].self, from: data)
                DispatchQueue.main.async { contacts = decoded }
            } catch {
                if let errorDict = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
                   let serverError = errorDict["error"] as? String {
                    DispatchQueue.main.async { errorMessage = "Server: \(serverError)" }
                } else {
                    DispatchQueue.main.async { errorMessage = "Failed to parse contacts." }
                }
            }
        }.resume()
    }

    private func updateGuardianStatus(contactId: Int, isGuardian: Bool, userId: Int) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/\(contactId)/guardian") else { return }
        var request = URLRequest(url: url)
        request.httpMethod = "PATCH"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try? JSONSerialization.data(withJSONObject: ["is_guardian": isGuardian, "userId": userId])
        URLSession.shared.dataTask(with: request).resume()
    }

    private func updatePrimaryStatus(contactId: Int, userId: Int) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/guardians/\(userId)/primary/\(contactId)") else { return }
        var request = URLRequest(url: url)
        request.httpMethod = "PUT"
        URLSession.shared.dataTask(with: request).resume()
    }

    private func uploadTrustedContact(name: String, phoneNumber: String, userId: Int, completion: @escaping (Bool) -> Void) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/bulk") else {
            completion(false)
            return
        }
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        
        let payload: [String: Any] = [
            "userId": userId,
            "contacts": [["name": name, "phone_numbers": [phoneNumber], "is_guardian": false]]
        ]
        request.httpBody = try? JSONSerialization.data(withJSONObject: payload)
        
        URLSession.shared.dataTask(with: request) { _, response, _ in
            DispatchQueue.main.async {
                let statusCode = (response as? HTTPURLResponse)?.statusCode ?? 0
                completion(statusCode == 200 || statusCode == 201)
            }
        }.resume()
    }
}

// MARK: - Native Contact Picker Wrapper
struct ContactPicker: UIViewControllerRepresentable {
    var onContactSelected: (String, String) -> Void
    @Environment(\.dismiss) private var dismiss

    func makeUIViewController(context: Context) -> CNContactPickerViewController {
        let picker = CNContactPickerViewController()
        picker.delegate = context.coordinator
        return picker
    }

    func updateUIViewController(_ uiViewController: CNContactPickerViewController, context: Context) {}

    func makeCoordinator() -> Coordinator { Coordinator(self) }

    class Coordinator: NSObject, CNContactPickerDelegate {
        var parent: ContactPicker
        init(_ parent: ContactPicker) { self.parent = parent }

        func contactPicker(_ picker: CNContactPickerViewController, didSelect contact: CNContact) {
            let name = "\(contact.givenName) \(contact.familyName)".trimmingCharacters(in: .whitespaces)
            let phone = contact.phoneNumbers.first?.value.stringValue ?? ""
            parent.onContactSelected(name, phone)
            parent.dismiss()
        }

        func contactPickerDidCancel(_ picker: CNContactPickerViewController) {
            parent.dismiss()
        }
    }
}

            }

            // 3. Call Screening Passcode Section

                header: Text("Owner Passcode"),
                footer: Text("Visitors must type the exact word or phrase to pass call screening.")
            ) {
                Button(action: {
                    showingPasscodeAlert = true
                }) {
                    HStack {
                        Image(systemName: "textformat")
                            .foregroundColor(.blue)

        .alert("Change Passcode Phrase", isPresented: $showingPasscodeAlert) {
            SecureField("Owner PIN", text: $passcodePinInput)

            TextField("New Word or Phrase", text: $newPasscodePhrase)
                .textInputAutocapitalization(.words)
            
            Button("Update Passcode") {
                updatePasscodeOnBackend(pin: passcodePinInput, passcode: newPasscodePhrase)

                newPasscodePhrase = ""

    private func validateAndUpdatePin() {
        guard !currentPinInput.isEmpty else {
            errorMessage = "Current PIN is required."
            clearPinInputs()
            return
        }
         
        guard newPinInput.count >= 4 else {
            errorMessage = "New PIN must be at least 4 digits."
            clearPinInputs()
            return
        }
         
        guard newPinInput == confirmNewPinInput else {
            errorMessage = "New PINs do not match."
            clearPinInputs()
            return
        }
         
        updatePinOnBackend(current: currentPinInput, new: newPinInput)
        clearPinInputs()
    }
     
    private func clearPinInputs() {
        currentPinInput = ""
        newPinInput = ""
        confirmNewPinInput = ""
    }

    private func updateShieldStatus(enabled: Bool, pin: String) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/user/toggle-shield") else { return }
         
        var request = URLRequest(url: url)
        request.httpMethod = "PUT"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
         
        let body: [String: Any] = [
            "userId": loggedInUserId,
            "isEnabled": enabled,
            "pin": pin
        ]
        request.httpBody = try? JSONSerialization.data(withJSONObject: body)

        URLSession.shared.dataTask(with: request) { _, response, _ in
            DispatchQueue.main.async {
                if let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 200 {
                    isShieldEnabled = enabled
                    successMessage = "Shield status updated successfully."
                    errorMessage = nil
                } else {
                    isShieldEnabled = !enabled
                    errorMessage = "Incorrect PIN or server error."
                    successMessage = nil
                }
            }
        }.resume()
    }
     
    private func updatePinOnBackend(current: String, new: String) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/user/update-pin") else { return }

ContactSyncManager

import Foundation
import Contacts
import Combine

class ContactSyncManager: ObservableObject {
    @Published var syncedContacts: [MCSContactModel] = []
    @Published var unsyncedContacts: [MCSContactModel] = []
    
    // MARK: - Fetch Local Device Contacts & Compare with Backend
    func fetchDeviceContacts() {
        let store = CNContactStore()
        
        store.requestAccess(for: .contacts) { [weak self] granted, error in
            guard granted, let self = self else { return }
            
            let keysToFetch = [
                CNContactGivenNameKey,
                CNContactFamilyNameKey,
                CNContactPhoneNumbersKey
            ] as [CNKeyDescriptor]
            
            let request = CNContactFetchRequest(keysToFetch: keysToFetch)
            var allDeviceContacts: [MCSContactModel] = []
            
            do {
                try store.enumerateContacts(with: request) { contact, _ in
                    let fullName = "\(contact.givenName) \(contact.familyName)".trimmingCharacters(in: .whitespaces)
                    let contactName = fullName.isEmpty ? "Unknown Contact" : fullName
                    
                    for phoneNumberObj in contact.phoneNumbers {
                        let rawNumber = phoneNumberObj.value.stringValue
                        let normalizedNumber = rawNumber.components(separatedBy: CharacterSet.decimalDigits.inverted).joined()
                        
                        if !normalizedNumber.isEmpty {
                            let model = MCSContactModel(name: contactName, phoneNumber: normalizedNumber, isTrusted: false, isPrimary: false)
                            if !allDeviceContacts.contains(where: { $0.phoneNumber == normalizedNumber }) {
                                allDeviceContacts.append(model)
                            }
                        }
                    }
                }
                
                DispatchQueue.main.async {
                    self.unsyncedContacts = allDeviceContacts
                }
            } catch {
                print("[ERROR] Failed to fetch local device contacts: \(error)")
            }
        }
    }
    
    // MARK: - Fetch Backend Contacts for Comparison
    func fetchBackendContacts(userId: Int) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/\(userId)") else { return }
        
        URLSession.shared.dataTask(with: url) { [weak self] data, response, error in
            guard let self = self, let data = data, error == nil else {
                print("[ERROR] Network error: \(String(describing: error))")
                return
            }
            
            // Check HTTP status code
            if let httpResponse = response as? HTTPURLResponse {
                print("[DEBUG] Status code: \(httpResponse.statusCode)")
                
                if !(200...299).contains(httpResponse.statusCode) {
                    if let errorString = String(data: data, encoding: .utf8) {
                        print("[ERROR] Server returned error payload: \(errorString)")
                    }
                    return
                }
            }
            
            if let jsonString = String(data: data, encoding: .utf8) {
                print("[DEBUG] Backend raw response: \(jsonString)")
            }
            
            struct BackendContact: Codable {
                let name: String
                let phone_number: String
                let is_guardian: Bool?
                let is_primary: Bool?
            }
            
            do {
                let backendList = try JSONDecoder().decode([BackendContact].self, from: data)
                let backendNumbers = Set(backendList.map { $0.phone_number })
                
                DispatchQueue.main.async {
                    var synced: [MCSContactModel] = []
                    var unsynced: [MCSContactModel] = []
                    
                    let combinedCurrent = self.unsyncedContacts + self.syncedContacts
                    
                    for contact in combinedCurrent {
                        let isBackendMatch = backendNumbers.contains(contact.phoneNumber)
                        if isBackendMatch {
                            let matchInfo = backendList.first(where: { $0.phone_number == contact.phoneNumber })
                            let updated = MCSContactModel(
                                name: contact.name,
                                phoneNumber: contact.phoneNumber,
                                isTrusted: matchInfo?.is_guardian ?? true,
                                isPrimary: matchInfo?.is_primary ?? false
                            )
                            synced.append(updated)
                        } else {
                            unsynced.append(contact)
                        }
                    }
                    
                    if combinedCurrent.isEmpty {
                        for bc in backendList {
                            synced.append(MCSContactModel(
                                name: bc.name,
                                phoneNumber: bc.phone_number,
                                isTrusted: bc.is_guardian ?? true,
                                isPrimary: bc.is_primary ?? false
                            ))
                        }
                    }
                    
                    self.syncedContacts = synced
                    self.unsyncedContacts = unsynced
                }
            } catch {
                print("[ERROR] Failed to parse contact list: \(error)")
            }
        }.resume()
    }
    
    /// Requests access to device contacts, fetches them, and uploads them to the backend in bulk.
    static func syncDeviceContacts(userId: Int, completion: @escaping (Bool, String) -> Void) {
        let store = CNContactStore()
        
        store.requestAccess(for: .contacts) { granted, error in
            guard granted else {
                DispatchQueue.main.async {
                    completion(false, "Permission to access contacts was denied.")
                }
                return
            }
            
            let keysToFetch = [
                CNContactGivenNameKey,
                CNContactFamilyNameKey,
                CNContactPhoneNumbersKey
            ] as [CNKeyDescriptor]
            
            let request = CNContactFetchRequest(keysToFetch: keysToFetch)
            var formattedContacts: [[String: Any]] = []
            
            do {
                try store.enumerateContacts(with: request) { contact, _ in
                    let fullName = "\(contact.givenName) \(contact.familyName)".trimmingCharacters(in: .whitespaces)
                    let contactName = fullName.isEmpty ? "Unknown Contact" : fullName
                    
                    var phoneNumbers: [String] = []
                    for phoneNumberObj in contact.phoneNumbers {
                        let rawNumber = phoneNumberObj.value.stringValue
                        let normalizedNumber = rawNumber.components(separatedBy: CharacterSet.decimalDigits.inverted).joined()
                        
                        if !normalizedNumber.isEmpty && !phoneNumbers.contains(normalizedNumber) {
                            phoneNumbers.append(normalizedNumber)
                        }
                    }
                    
                    if !phoneNumbers.isEmpty {
                        let contactDict: [String: Any] = [
                            "name": contactName,
                            "phone_numbers": phoneNumbers,
                            "is_guardian": false
                        ]
                        formattedContacts.append(contactDict)
                    }
                }
                
                sendBulkContactsToServer(userId: userId, contacts: formattedContacts, completion: completion)
                
            } catch {
                DispatchQueue.main.async {
                    completion(false, "Failed to fetch device contacts: \(error.localizedDescription)")
                }
            }
        }
    }
    
    private static func sendBulkContactsToServer(userId: Int, contacts: [[String: Any]], completion: @escaping (Bool, String) -> Void) {
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/bulk") else {
            DispatchQueue.main.async { completion(false, "Invalid server URL.") }
            return
        }
        
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        
        let payload: [String: Any] = [
            "userId": userId,
            "contacts": contacts
        ]
        
        do {
            request.httpBody = try JSONSerialization.data(withJSONObject: payload, options: [])
        } catch {
            DispatchQueue.main.async { completion(false, "Failed to encode contacts payload.") }
            return
        }
        
        URLSession.shared.dataTask(with: request) { _, response, error in
            DispatchQueue.main.async {
                if let error = error {
                    completion(false, "Network error: \(error.localizedDescription)")
                    return
                }
                
                guard let httpResponse = response as? HTTPURLResponse, (200...299).contains(httpResponse.statusCode) else {
                    completion(false, "Server rejected the bulk sync request.")
                    return
                }
                
                completion(true, "Contacts synced successfully!")
            }
        }.resume()
    }
    
    func syncAllContacts(userId: Int) {
        print("DEBUG: 'Sync All' button pressed. Requesting contact access...")
        
        let store = CNContactStore()
        
        store.requestAccess(for: .contacts) { [weak self] granted, error in
            guard let self = self else { return }
            
            if let error = error {
                print("DEBUG ERROR: Permission request returned error: \(error.localizedDescription)")
                return
            }
            
            guard granted else {
                print("DEBUG WARNING: User denied contacts permission.")
                return
            }
            
            print("DEBUG: Permission granted. Starting contact fetch...")
            
            let keysToFetch = [
                CNContactGivenNameKey,
                CNContactFamilyNameKey,
                CNContactPhoneNumbersKey
            ] as [CNKeyDescriptor]
            
            let request = CNContactFetchRequest(keysToFetch: keysToFetch)
            var formattedContacts: [[String: Any]] = []
            
            do {
                try store.enumerateContacts(with: request) { contact, _ in
                    let fullName = "\(contact.givenName) \(contact.familyName)".trimmingCharacters(in: .whitespaces)
                    let contactName = fullName.isEmpty ? "Unknown Contact" : fullName
                    
                    var phoneNumbers: [String] = []
                    for phoneNumberObj in contact.phoneNumbers {
                        let rawNumber = phoneNumberObj.value.stringValue
                        let normalizedNumber = rawNumber.components(separatedBy: CharacterSet.decimalDigits.inverted).joined()
                        
                        if !normalizedNumber.isEmpty && !phoneNumbers.contains(normalizedNumber) {
                            phoneNumbers.append(normalizedNumber)
                        }
                    }
                    
                    if !phoneNumbers.isEmpty {
                        let contactDict: [String: Any] = [
                            "name": contactName,
                            "phone_numbers": phoneNumbers,
                            "is_guardian": false
                        ]
                        formattedContacts.append(contactDict)
                    }
                }
                
                print("DEBUG: Fetched \(formattedContacts.count) contacts. Uploading to server...")
                
                ContactSyncManager.sendBulkContactsToServer(userId: userId, contacts: formattedContacts) { success, message in
                    print("DEBUG: Bulk upload response - Success: \(success), Message: \(message)")
                    if success {
                        self.fetchDeviceContacts()
                        self.fetchBackendContacts(userId: userId)
                    }
                }
                
            } catch {
                print("DEBUG CRITICAL ERROR: Failed to fetch contacts: \(error.localizedDescription)")
            }
        }
    }
    
    // MARK: - Save Single Contact
    func saveSingleContact(userId: Int, name: String, phoneNumber: String, completion: (() -> Void)? = nil) {
        guard userId > 0 else {
            print("ERROR: Invalid userId")
            return
        }
        
        let payload: [String: Any] = [
            "userId": userId,
            "contacts": [
                [
                    "name": name,
                    "phone_numbers": [phoneNumber],
                ]
            ]
        ]
        
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/trusted-contacts/bulk") else { return }
        
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        
        do {
            request.httpBody = try JSONSerialization.data(withJSONObject: payload)
        } catch {
            print("ERROR: Failed to encode single contact payload")
            return
        }
        
        URLSession.shared.dataTask(with: request) { [weak self] data, response, error in
            if let error = error {
                print("ERROR: Failed to save single contact: \(error.localizedDescription)")
                return
            }
            
            DispatchQueue.main.async {
                self?.fetchBackendContacts(userId: userId)
                completion?()
            }
        }.resume()
    }
}

GuardianService

import Foundation
import SwiftUI
import Combine

// MARK: - Models
struct Guardian: Identifiable, Codable {
    let id: Int?
    let userId: Int?
    let guardian_name: String
    let guardian_phone: String

    enum CodingKeys: String, CodingKey {
        case id
        case userId = "user_id"
        case guardian_name
        case guardian_phone
    }
}

// MARK: - Network Manager for Guardians / Trusted Contacts
class GuardianService: ObservableObject {
    @Published var guardians: [Guardian] = []
    @Published var isLoading: Bool = false
    @Published var errorMessage: String?

    private let baseURL = "https://mobilecallshield-backend.onrender.com/api/guardians"

    // 1. Fetch Guardians for User
    func fetchGuardians(userId: Int) {
        isLoading = true
        guard let url = URL(string: "\(baseURL)/\(userId)") else {
            isLoading = false
            return
        }

        URLSession.shared.dataTask(with: url) { [weak self] data, response, error in
            DispatchQueue.main.async {
                self?.isLoading = false
                if let error = error {
                    self?.errorMessage = error.localizedDescription
                    print("❌ Fetch error: \(error)")
                    return
                }

                guard let data = data else { return }
                
                // Debug response
                if let jsonString = String(data: data, encoding: .utf8) {
                    print("📡 Guardians Response: \(jsonString)")
                }

                do {
                    let decoded = try JSONDecoder().decode([Guardian].self, from: data)
                    self?.guardians = decoded
                } catch {
                    print("❌ Decode error: \(error)")
                    self?.errorMessage = error.localizedDescription
                }
            }
        }.resume()
    }

    // 2. Add a Trusted Contact / Guardian
    func addGuardian(userId: Int, name: String, phone: String, completion: @escaping (Bool) -> Void) {
        guard let url = URL(string: baseURL) else {
            completion(false)
            return
        }

        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")

        let body: [String: Any] = [
            "userId": userId,
            "guardian_name": name,
            "guardian_phone": phone
        ]

        do {
            request.httpBody = try JSONSerialization.data(withJSONObject: body)
        } catch {
            print("❌ Request serialization error: \(error)")
            completion(false)
            return
        }

        URLSession.shared.dataTask(with: request) { [weak self] data, response, error in
            DispatchQueue.main.async {
                if let error = error {
                    print("❌ Add error: \(error)")
                    completion(false)
                    return
                }

                if let httpResponse = response as? HTTPURLResponse {
                    print("📡 Add Status Code: \(httpResponse.statusCode)")
                    if httpResponse.statusCode == 200 || httpResponse.statusCode == 201 {
                        // Refresh the list after adding
                        self?.fetchGuardians(userId: userId)
                        completion(true)
                        return
                    }
                }
                completion(false)
            }
        }.resume()
    }

    // 3. Delete a Trusted Contact / Guardian
    func deleteGuardian(id: Int, userId: Int, completion: @escaping (Bool) -> Void) {
        guard let url = URL(string: "\(baseURL)/\(id)") else {
            completion(false)
            return
        }

        var request = URLRequest(url: url)
        request.httpMethod = "DELETE"

        URLSession.shared.dataTask(with: request) { [weak self] data, response, error in
            DispatchQueue.main.async {
                if let error = error {
                    print("❌ Delete error: \(error)")
                    completion(false)
                    return
                }

                if let httpResponse = response as? HTTPURLResponse {
                    print("📡 Delete Status Code: \(httpResponse.statusCode)")
                    if httpResponse.statusCode == 200 {
                        // Refresh the list after deleting
                        self?.fetchGuardians(userId: userId)
                        completion(true)
                        return
                    }
                }
                completion(false)
            }
        }.resume()
    }
}

    private func performLogin() {
        isLoading = true
        errorMessage = ""
         
        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/auth/login") else {
            isLoading = false
            errorMessage = "Invalid server URL."
            return
        }
         
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
         
        let body: [String: Any] = [
            "email": username.trimmingCharacters(in: .whitespaces),
            "password": password
        ]
         
        request.httpBody = try? JSONSerialization.data(withJSONObject: body)
         
        URLSession.shared.dataTask(with: request) { data, response, error in
            DispatchQueue.main.async {
                isLoading = false
                 
                if let error = error {
                    errorMessage = "Network error: \(error.localizedDescription)"
                    return
                }
                 
                guard let data = data else {
                    errorMessage = "No response from server."
                    return
                }
                 
                do {
                    if let json = try JSONSerialization.jsonObject(with: data) as? [String: Any] {
                        let successFlag = json["success"] as? Bool ?? false
                         
                        if successFlag {
                            let resolvedUserId: Int
                            if let userDict = json["user"] as? [String: Any],

                                userRole = (userDict["role"] as? String) ?? "user"

                        } else {
                            let serverError = json["error"] as? String ?? "Unknown server rejection."
                            errorMessage = serverError

                } catch {

MobileCarrier

import Foundation

enum MobileCarrier: String, CaseIterable, Identifiable, Codable {
    case att = "AT&T"
    case verizon = "Verizon"
    case tmobile = "T-Mobile"
    case visible = "Visible (Verizon)"
    case totalWireless = "Total Wireless (Verizon)"
    case straightTalk = "Straight Talk"
    case cricket = "Cricket Wireless (AT&T)"
    case mint = "Mint Mobile (T-Mobile)"
    case metro = "Metro by T-Mobile"
    
    var id: String { self.rawValue }
    
    /// Generates carrier-specific USSD star code to activate conditional call forwarding
    func forwardingCode(for twilioNumber: String) -> String {
        // Strip out non-numeric characters (leaving only digits)
        let digitsOnly = twilioNumber.components(separatedBy: CharacterSet.decimalDigits.inverted).joined()
        
        // Take the last 10 digits (e.g., 2812476667)
        let tenDigitNumber = String(digitsOnly.suffix(10))
        
        switch self {
        case .verizon, .visible, .totalWireless, .straightTalk:
            // Verizon network code format (*71 + 10 digits)
            return "*71\(tenDigitNumber)"
            
        case .att, .cricket:
            // AT&T network code format (*61* + 10 digits + #)
            return "*61*\(tenDigitNumber)#"
            
        case .tmobile, .mint, .metro:
            // T-Mobile network code format (**61* + 10 digits + **15#)
            return "**61*\(tenDigitNumber)**15#"
        }
    }
    
    /// Generates the carrier code to disable call forwarding
    var disableCode: String {
        switch self {
        case .verizon, .visible, .totalWireless, .straightTalk:
            return "*73" //
        case .att, .cricket, .tmobile, .mint, .metro:
            return "##61#" //
        }
    }
}

            isLoading = false
            return
        }

        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")

        let body: [String: Any] = [
            "userId": loggedInUserId,
            "areaCode": areaCodeInput
        ]
        request.httpBody = try? JSONSerialization.data(withJSONObject: body)

        URLSession.shared.dataTask(with: request) { data, response, error in
            DispatchQueue.main.async {
                isLoading = false
                if let error = error {
                    errorMessage = "Network error: \(error.localizedDescription)"
                    return
                }

                guard let data = data, let httpRes = response as? HTTPURLResponse else {
                    errorMessage = "Invalid response from server."
                    return
                }

                do {
                    if let json = try JSONSerialization.jsonObject(with: data) as? [String: Any],
                       let success = json["success"] as? Bool, success {
                        self.assignedNumber = json["assignedPhoneNumber"] as? String
                        successMessage = "Phone number successfully assigned!"
                    } else {
                        let errStr = (try? JSONSerialization.jsonObject(with: data) as? [String: Any])?["error"] as? String
                        errorMessage = errStr ?? "Failed to provision number."
                    }
                } catch {
                    errorMessage = "Failed to decode server response."
                }
            }
        }.resume()
    }

// MARK: - Address Autocomplete Search Manager
@MainActor
class AddressSearchManager: ObservableObject {
    @Published var searchQuery: String = ""
    @Published var searchResults: [MKMapItem] = []
    
    private var searchCancellable: AnyCancellable?
    
    init() {
        searchCancellable = $searchQuery
            .debounce(for: .milliseconds(300), scheduler: RunLoop.main)
            .removeDuplicates()
            .sink { [weak self] query in
                self?.performSearch(query: query)
            }
    }
    
    private func performSearch(query: String) {
        let trimmed = query.trimmingCharacters(in: .whitespaces)
        guard !trimmed.isEmpty, trimmed.count > 2 else {
            self.searchResults = []
            return
        }
         
        let request = MKLocalSearch.Request()
        request.naturalLanguageQuery = trimmed
        request.resultTypes = .address
         
        let search = MKLocalSearch(request: request)
        search.start { response, error in
            guard let response = response, error == nil else { return }
            Task { @MainActor in
                self.searchResults = Array(response.mapItems.prefix(4))
            }
        }
    }
}

        var formattedPhone = phoneNumber.components(separatedBy: CharacterSet.decimalDigits.inverted).joined()
        if formattedPhone.count == 10 {
            formattedPhone = "+1" + formattedPhone
        } else if !formattedPhone.hasPrefix("+") {
            formattedPhone = "+" + formattedPhone
        }

        guard let url = URL(string: "https://mobilecallshield-backend.onrender.com/api/auth/register") else {
            isLoading = false
            errorMessage = "Invalid server URL."
            return
        }

        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")

        let body: [String: Any] = [
            "name": name.trimmingCharacters(in: .whitespaces),
            "phoneNumber": formattedPhone,
            "email": email.trimmingCharacters(in: .whitespaces),
            "password": password,
            "streetAddress": streetAddress.trimmingCharacters(in: .whitespaces),
            "city": city.trimmingCharacters(in: .whitespaces),
            "state": standardizeState(state),
            "zipCode": zipCode.trimmingCharacters(in: .whitespaces),
            "country": country.trimmingCharacters(in: .whitespaces),
            "pin": pin.trimmingCharacters(in: .whitespaces),
            "passcode": passcode.trimmingCharacters(in: .whitespaces)
        ]

        request.httpBody = try? JSONSerialization.data(withJSONObject: body)

        URLSession.shared.dataTask(with: request) { data, response, error in
            DispatchQueue.main.async {
                isLoading = false

                if let error = error {
                    errorMessage = "Network error: \(error.localizedDescription)"
                    return
                }

                guard let data = data else {
                    errorMessage = "No response from server."
                    return
                }

                do {
                    if let json = try JSONSerialization.jsonObject(with: data) as? [String: Any] {
                        let successFlag = json["success"] as? Bool ?? false
                        if successFlag {
                            successMessage = "Account created successfully! You can now log in."
                            DispatchQueue.main.asyncAfter(deadline: .now() + 1.5) {
                                dismiss()
                            }
                        } else {
                            errorMessage = json["error"] as? String ?? "Registration failed."
                        }
                    }
                } catch {
                    errorMessage = "Failed to parse server response."
                }
            }
        }.resume()
    }
}

        URLSession.shared.dataTask(with: request) { data, _, error in

                        if success {
                            isGuardianShieldActive = isEnabled
                        }

        }.resume()
    }
